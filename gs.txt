// ============================================================
// 📧 GMAIL SEARCH FILTERS
// ============================================================
const GMAIL_SEARCH_QUERY = 'subject:("Enquiry for" OR "Requirement" OR "Contact") is:unread';

// ============================================================
// 🏷️ TAG CONSTANTS
// ============================================================
const LEAD_SOURCES = {
  INDIAMART: "indiamart",
  MAPS: "google-maps",
  MANUAL: "manual",
  FACEBOOK: "facebook"
};

// ============================================================
// Firebase Connection Helper
// ============================================================
function getFirestore() {
  return FirestoreApp.getFirestore(
    FIREBASE_SERVICE_ACCOUNT_EMAIL,
    FIREBASE_PRIVATE_KEY,
    FIREBASE_PROJECT_ID
  );
}

/**
 * Creates a lead/contact record in Firestore.
 * Used by background scrapers to save normalized lead contacts.
 *
 * @param {Object} leadData - Lead payload values.
 * @returns {Object} created lead record.
 */
function createLeadContact(leadData) {
  const firestore = getFirestore();
  const now = new Date();
  const payload = leadData || {};

  const newLead = {
    id: payload.id || ('LD-' + now.getTime() + Math.floor(Math.random() * 1000)),
    mobile: payload.mobile || '',
    name: payload.name || 'Unknown Lead',
    source: payload.source || LEAD_SOURCES.MANUAL,
    Tag: payload.Tag || payload.tag || null,
    status: payload.status || 'New',
    createdAt: payload.createdAt || now.toISOString(),
    createdDate: payload.createdDate || now.toISOString().split('T')[0],
    address: payload.address || 'Not Available',
    product: payload.product || 'General Enquiry',
    raw: payload.raw || '',
    notes: payload.notes || ''
  };

  firestore.createDocument('leads/' + newLead.id, newLead);
  return newLead;
}

// 📧 1. SCRAPE INDIAMART & PUSH TO FIREBASE (Original Logic)
// ============================================================
function processIndiaMartEmailsToFirebase() {
  console.log("Starting IndiaMART Email Processor...");
  
  const threads = GmailApp.search(GMAIL_SEARCH_QUERY);
  let count = 0;
  let sessionNumbers = new Set();

  threads.forEach(thread => {
    const m = thread.getMessages()[0]; // First message only

    try {
      const plainBody = m.getPlainBody();
      const htmlBody  = m.getBody();
      const subject   = m.getSubject();
      const massiveBlock = plainBody + " " + htmlBody;

      let validMob = null;
      const rawMatches = massiveBlock.match(/\d{10,13}/g) || [];

      for (let raw of rawMatches) {
        const clean = raw.slice(-10);
        if (!['6','7','8','9'].includes(clean[0])) continue;
        if (clean.startsWith("800") || clean.startsWith("1800")) continue;
        if (sessionNumbers.has(clean)) { validMob = null; break; }
        validMob = clean;
        break;
      }

      if (validMob) {
        sessionNumbers.add(validMob);

        const lines = plainBody.split('\n').map(l => l.trim()).filter(l => l.length > 0);
        let extractedName = "Unknown";
        let loc = "";

        const regardsIdx = lines.findIndex(l => l.toLowerCase().startsWith("regards"));
        if (regardsIdx !== -1 && regardsIdx + 1 < lines.length) {
          extractedName = lines[regardsIdx + 1];
          for (let i = regardsIdx + 2; i < regardsIdx + 8; i++) {
            if (i >= lines.length) break;
            const line = lines[i];
            const low  = line.toLowerCase();
            if (low.includes("@") || low.includes("mobile") || low.includes("call")) continue;
            if (line.replace(/\D/g,'').includes(validMob)) continue;
            if (low.includes("member since") || low.includes("gst") || low.includes("verified") || line.length < 3) continue;
            loc = line;
            break;
          }
        }

        const prodMatch = subject.match(/Enquiry for\s+(.+?)(\s+from|$)/i);
        const product   = prodMatch ? prodMatch[1].trim() : "General Enquiry";
        const finalName = extractedName !== "Unknown" ? extractedName : "IndiaMART Lead";

        const newLead = createLeadContact({
          id: 'LD-IM-' + Date.now() + Math.floor(Math.random() * 1000),
          mobile: validMob,
          name: finalName,
          source: LEAD_SOURCES.INDIAMART,
          Tag: null,
          status: 'New',
          createdAt: new Date().toISOString(),
          createdDate: new Date().toISOString().split('T')[0],
          address: loc || "Not Available",
          product: product,
          raw: `${finalName} | ${loc} | ${product} | ${validMob}`,
          notes: `Name: ${finalName}\nAddr: ${loc}\nProduct: ${product}`
        });

        console.log(`✅ Saved Lead to Firebase: ${finalName} (${validMob}) | Source: ${newLead.source}`);

        count++;
        thread.markRead();
      }
    } catch (e) {
      console.log("Email Error: " + e.message);
    }
  });
  
  console.log(`Finished. Processed ${count} unique leads directly to Firebase.`);
}

// 🗺️ 2. HUNT GOOGLE MAPS LEADS & PUSH TO FIREBASE
// ============================================================
function huntDailyLeadsToFirebase() {
  console.log("Starting Daily Map Hunt for Firebase...");
  const props = PropertiesService.getScriptProperties();

  let areaIdx = Number(props.getProperty('LAST_AREA_IDX') || 0);
  let keyIdx  = Number(props.getProperty('LAST_KEYWORD_IDX') || 0);

  const area = TARGET_AREAS[areaIdx];
  const keyword = TARGET_KEYWORDS[keyIdx];
  const query = `${keyword} in ${area} ${TARGET_CITY}`;

  const options = {
    method: "post",
    contentType: "application/json",
    headers: {
      "X-Goog-Api-Key": MAPS_KEY,
      "X-Goog-FieldMask": "places.displayName,places.formattedAddress,places.rating,places.nationalPhoneNumber"
    },
    payload: JSON.stringify({ textQuery: query }),
    muteHttpExceptions: true
  };

  try {
    const json = JSON.parse(UrlFetchApp.fetch("https://places.googleapis.com/v1/places:searchText", options).getContentText());
    if (json.error) return console.log("API Error: " + json.error.message);

    let saved = 0;
    for (let place of (json.places || [])) {
      if (saved >= MAPS_DAILY_LIMIT) break;

      const phone = place.nationalPhoneNumber;
      if (!phone) continue;
      const cleanPhone = phone.replace(/[^0-9]/g, '').slice(-10);
      if (cleanPhone.length < 10) continue;

      const name   = place.displayName ? place.displayName.text : "Unknown";
      const addr   = place.formattedAddress || TARGET_CITY;
      const rating = place.rating || null;

      // Save hunted lead as contact-style entry with fixed Facebook name
      const newLead = createLeadContact({
        id: 'LD-GM-' + Date.now() + Math.floor(Math.random() * 1000),
        mobile: cleanPhone,
        name: 'Facebook Lead',
        source: LEAD_SOURCES.FACEBOOK,
        Tag: null,
        status: 'New',
        createdAt: new Date().toISOString(),
        createdDate: new Date().toISOString().split('T')[0],
        address: addr,
        product: keyword,
        raw: `Facebook Lead | ${addr} | ${cleanPhone}`,
        notes: `Source: Google Maps\nOriginal Name: ${name}\nArea: ${area}\nType: ${keyword}\nRating: ${rating || "N/A"}`
      });

      saved++;
      console.log(`✅ Saved Lead to Firebase: ${name} (${cleanPhone}) | Source: ${newLead.source}`);
    }

    // Rotate targets
    areaIdx++;
    if (areaIdx >= TARGET_AREAS.length) { areaIdx = 0; keyIdx++; }
    if (keyIdx >= TARGET_KEYWORDS.length) keyIdx = 0;
    props.setProperty('LAST_AREA_IDX', areaIdx);
    props.setProperty('LAST_KEYWORD_IDX', keyIdx);
    
    console.log(`Map Hunt Complete. Pushed ${saved} leads to Firebase.`);
  } catch (e) {
    console.log("Hunt Error: " + e.message);
  }
}

// 👤 3. CREATE CONTACT FROM LEAD (FACEBOOK FIXED NAME)
// ============================================================
/**
 * Adds a new lead as a contact-style record in Firestore.
 * Name is always fixed to "Facebook Lead".
 *
 * @param {string} mobile - Lead mobile number.
 * @param {Object} extraData - Optional extra fields (address, notes, product, etc.)
 * @returns {Object} created lead/contact record or error payload.
 */
function addFacebookLeadContact(mobile, extraData) {
  const cleanMobile = String(mobile || '').replace(/\D/g, '').slice(-10);
  if (cleanMobile.length !== 10) {
    return { error: 'Valid 10-digit mobile is required.' };
  }

  const payload = extraData || {};

  const newLead = createLeadContact({
    id: 'LD-FB-' + Date.now() + Math.floor(Math.random() * 1000),
    mobile: cleanMobile,
    name: 'Facebook Lead',
    source: LEAD_SOURCES.FACEBOOK,
    Tag: payload.Tag || payload.tag || null,
    status: payload.status || 'New',
    address: payload.address || 'Not Available',
    product: payload.product || 'Facebook Enquiry',
    raw: `Facebook Lead | ${cleanMobile}`,
    notes: payload.notes || 'Added as Facebook Lead contact'
  });

  console.log(`✅ Saved Facebook Lead Contact: ${cleanMobile}`);
  return newLead;
}
